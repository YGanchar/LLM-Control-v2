Use EXACT project structure:

LLM-Control-v2/
├── .env                    # Конфигурация (пути, SSH-пароли, MAC-адрес)
├── main.py                 # Точка входа приложения
├── main_ui.py              # Главное окно (координация виджетов)
├── scanner_widget.py       # Виджет сканирования моделей
├── config_widget.py        # Виджет конфигурации .mod файлов
├── server_widget.py        # Виджет управления сервером
├── model_layers.json       # Маппинг размеров моделей → количество слоев (n_layer)
├── services/
│   ├── model_scanner.py    # Сканирование файловой системы
│   ├── system_monitor.py   # Мониторинг удалённого сервера по SSH
│   ├── server_control.py   # Локальный контроль процессов
│   └── mod_generator.py    # Генератор .mod файлов (справочный)
└── dist/
